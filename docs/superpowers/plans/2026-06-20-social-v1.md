# Social v1 — friends, activity feed, kudos — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the between-class social layer for champ-app members — a mutual-friends graph, a compute-on-read activity feed, tap reactions (kudos), and a friends-only leaderboard.

**Architecture:** All member surfaces + social APIs live in **champ-app** (customer-facing). The shared Supabase project gets one migration (`301`) for `member_friendships` + `feed_reactions` + `member_social_settings` and a widened `customer_engagement_nudges` CHECK. Pure helpers (`social.js`) are byte-synced un1t-crm↔champ-app like `challenges.js`; the DB reads/aggregates are champ-app-side IO (`load-social.js`) using the service client, mirroring `load-challenges.js`. The pure push **builders** (`customer-notifications.js`) become byte-synced; the Expo **sender** is a per-repo IO port (`customer-push.js`) into champ-app. A per-location kill-switch lives in `locations.settings.social_enabled`.

**Tech Stack:** Next.js 16 (champ-app web), Expo/React Native (champ-app mobile), Supabase (shared project, RLS + service client), Vitest, Expo Push.

**Spec:** `docs/superpowers/specs/2026-06-20-social-v1-design.md`. Locked decisions: full loop (friends + boards + feed + reactions); mutual friends (request→accept) + shared-class suggestions; summary-only visibility (no raw HR); tap-only preset reactions; same-studio v1; compute-on-read feed.

```js
// Social v1 pure helpers — friend-status derivation, feed merge, reaction summary,
// suggestion ranking. No IO. Byte-synced to champ-app/shared/social.js.

export const REACTIONS = [
  { key: 'strong', emoji: '💪', label: 'Strong' },
  { key: 'fire', emoji: '🔥', label: 'Fire' },
  { key: 'clap', emoji: '👏', label: 'Clap' },
  { key: 'wow', emoji: '😮', label: 'Wow' },
]
const REACTION_KEYS = REACTIONS.map((r) => r.key)

export function friendshipPairKey(a, b) {
  return [a, b].sort().join(':')
}

export function friendStatusFor(row, myContactId) {
  if (!row) return null
  if (row.status === 'blocked') return 'blocked'
  if (row.status === 'accepted') return 'friends'
  // pending
  return row.requester_contact_id === myContactId ? 'outgoing' : 'incoming'
}

export function mergeFeed(items) {
  return [...(items || [])].sort((a, b) => (b.ts || 0) - (a.ts || 0))
}

export function reactionSummary(rows, myContactId) {
  const counts = Object.fromEntries(REACTION_KEYS.map((k) => [k, 0]))
  let mine = null
  for (const r of rows || []) {
    if (counts[r.reaction] !== undefined) counts[r.reaction] += 1
    if (r.reactor_contact_id === myContactId) mine = r.reaction
  }
  const total = REACTION_KEYS.reduce((a, k) => a + counts[k], 0)
  return { counts, total, mine }
}

export function rankSuggestions(rows) {
  return [...(rows || [])].sort(
    (a, b) => (b.sharedClasses || 0) - (a.sharedClasses || 0) || String(a.name || '').localeCompare(String(b.name || ''))
  )
}
```

```js
// champ-app social IO — friends-board standings (and, later tasks, feed + suggestions).
// Service client only (cross-member). Pure helpers from ./challenges + ./social.
import { metricValue, rankStandings, shortName } from './challenges.js'
import { logWarn } from './log.js'

const PAGE = 1000, HARD_LIMIT = 20000

async function windowSessionsFor(db, { locationId, fromIso, toIso, contactIds }) {
  if (!contactIds?.length) return []
  const out = []
  for (let from = 0; ; from += PAGE) {
    const { data, error } = await db
      .from('heart_rate_sessions')
      .select('contact_id, effort_points, zones_seconds, contacts!heart_rate_sessions_contact_id_fkey(name)')
      .eq('location_id', locationId)
      .in('contact_id', contactIds)
      .not('ended_at', 'is', null)
      .gte('started_at', fromIso).lt('started_at', toIso)
      .order('contact_id', { ascending: true })
      .range(from, from + PAGE - 1)
    if (error) { logWarn('load-social', 'window read failed', { err: error, locationId }); break }
    out.push(...(data || []))
    if (!data || data.length < PAGE) break
    if (out.length >= HARD_LIMIT) break
  }
  return out
}

function standingsFrom(sessions, metric) {
  const byC = new Map()
  for (const s of sessions) {
    const cur = byC.get(s.contact_id) || { contactId: s.contact_id, value: 0, c: s.contacts }
    cur.value += metricValue(s, metric); byC.set(s.contact_id, cur)
  }
  return rankStandings([...byC.values()].map((r) => ({
    contactId: r.contactId, value: Math.round(r.value * 100) / 100, name: shortName(r.c?.name),
  })))
}

function startOfMonthIso(nowMs) {
  const d = new Date(nowMs)
  return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), 1)).toISOString()
}

export async function loadFriendsBoard(svc, { contactId, locationId, friendContactIds, nowMs = Date.now() }) {
  const ids = [...new Set([contactId, ...(friendContactIds || [])])]
  const sessions = await windowSessionsFor(svc, { locationId, fromIso: startOfMonthIso(nowMs), toIso: new Date(nowMs).toISOString(), contactIds: ids })
  const standings = standingsFrom(sessions, 'points')
  const mine = standings.find((r) => r.contactId === contactId) || null
  return {
    rows: standings.map((r) => ({ name: r.name, value: r.value, rank: r.rank, isMe: r.contactId === contactId })),
    me: mine ? { rank: mine.rank, value: mine.value } : null,
  }
}
```
